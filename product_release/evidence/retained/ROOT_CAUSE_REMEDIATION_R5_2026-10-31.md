```
import json
import os
import sys
import subprocess
from pathlib import Path

class InputValidationError(ValueError):
    pass

def require(condition, code):
    if not condition:
        raise InputValidationError(code)

def blocked(code):
    print("BLOCKED_SETUP: " + code, file=sys.stderr)
    raise SystemExit(78)

def object_value(value, code):
    require(type(value) is dict, code)
    return value

def member_object(parent, key, code):
    parent = object_value(parent, code)
    return object_value(parent[key], code)

def optional_object(parent, key, code):
    parent = object_value(parent, code)
    if key in parent:
        object_value(parent[key], code)

def json_file(path):
    raw = Path(path).read_bytes()
    return raw, json.loads(raw)

def main():
    try:
        inputs = read_binding_inputs(os.environ, sys.argv)
    except (
        OSError, UnicodeError, ImportError,
        KeyError, TypeError, ValueError,
        subprocess.SubprocessError,
    ) as exc:
        # Never echo private input contents or exception messages.
        blocked("INVALID_BINDING_INPUT:" + type(exc).__name__)

    emit_verified_inputs(inputs)

if __name__ == "__main__":
    main()
```
